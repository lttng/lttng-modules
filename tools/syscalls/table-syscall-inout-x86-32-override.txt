# On x86-32, the 64-bit mask parameter is split into two 32-bit parts
syscall fanotify_mark nbargs 6 rw: (r, r, r, r, r, r)
# OLD_SIGSUSPEND3: 3 args variant
syscall sigsuspend nbargs 3 rw: (r, r, r)
# CLONE_BACKWARDS: tls passed as the 4th argument of clone(2), not the 5th one.
syscall clone nbargs 5 rw: (r, r, w, r, w)
# sys_ia32_pread64: the 64-bit pos parameter is split into two 32-bit parts
syscall pread64 nbargs 5 rw: (r, w, r, r, r)
# sys_ia32_pwrite64: the 64-bit pos parameter is split into two 32-bit parts
syscall pwrite64 nbargs 5 rw: (r, r, r, r, r)
# sys_ia32_truncate64: the 64-bit offset parameter is split into two 32-bit parts
syscall truncate64 nbargs 3 rw: (r, r, r)
# sys_ia32_ftruncate64: the 64-bit offset parameter is split into two 32-bit parts
syscall ftruncate64 nbargs 3 rw: (r, r, r)
# sys_ia32_readahead: the 64-bit offset parameter is split into two 32-bit parts
syscall readahead nbargs 4 rw: (r, r, r, r)
# sys_ia32_fadvise64: the 64-bit offset parameter is split into two 32-bit parts
syscall fadvise64 nbargs 5 rw: (r, r, r, r, r)
# sys_ia32_fadvise64_64: the 64-bit offset and len parameters are split into two 32-bit parts
syscall fadvise64_64 nbargs 6 rw: (r, r, r, r, r, r)
# sys_ia32_sync_file_range: the 64-bit offset and nbytes parameters are split into two 32-bit parts
syscall sync_file_range nbargs 6 rw: (r, r, r, r, r, r)
# sys_ia32_fallocate: the 64-bit offset and len parameters are split into two 32-bit parts
syscall fallocate nbargs 6 rw: (r, r, r, r, r, r)
